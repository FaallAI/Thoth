### Task 2: Seroval decoder

**Files:**
- Create: `python/src/thoth_control_plane/acquisition/adapters/seroval.py`
- Test: `python/tests/acquisition/test_seroval.py`

**Interfaces:**
- Consumes: nothing.
- Produces: `decode_scripts(html: str) -> list[Any]`. Each value is one decoded payload,
  with `$R[n]` references resolved per stream key. Values are dicts, lists, `str`, `int`,
  `float`, `bool`, or `None`. They may be cyclic.

The grammar comes from the probe pages. Data payloads look like this:

```text
($R=>$R[74].next($R[77]={...}))($R["<key>"])
```

- Inside a payload: bare keys, `"..."` strings with `\"` escapes, `!0` and `!1`, `void 0`,
  `$R[n]=` assignments, and `$R[n]` references.
- The first script holds the router runtime, which contains functions. It must decode to
  nothing.
- `.return(void 0)` wrappers end a stream and carry no data.

from __future__ import annotations

from thoth_control_plane.acquisition.adapters.seroval import decode_scripts


def _page(*scripts: str) -> str:
    return "".join(f'<script nonce="n">{script}</script>' for script in scripts)


def _next(value: str, key: str = "k1", slot: int = 0) -> str:
    return f'($R=>$R[{slot}].next({value}))($R["{key}"]);document.currentScript.remove()'

        'l:[1,"\\x41\\u00e9"],o:{},e:[],"quoted key":1,7:"seven"}'
    )
    assert decode_scripts(_page(_next(value))) == [
        {
            "a": 'x"y\\z',
            "n": -12,
            "f": 150.0,
            "g": 2000.0,
            "t": True,
            "u": False,
            "v": None,

    first = _next('$R[1]={user:$R[2]={name:"a"},again:$R[2]}')
    second = _next("$R[3]={later:$R[2]}")

    assert two["later"] is one["user"]

```python
"""Decode the seroval-serialized data X's pages stream to the browser; no code is run.

Logged-out X pages stream their data through TanStack Router scripts of the form
`($R=>$R[n].next(<value>))($R["<key>"])`. `<value>` is a JavaScript literal: objects with
bare or quoted keys, arrays, strings, numbers, `!0`/`!1`, `void 0`, and `$R[n]=` assignments
and `$R[n]` references into a per-stream table. `decode_scripts` reads that subset only.
A wrapper holding anything else (the router runtime, functions, constructors) decodes to
nothing, and the decoder never raises.
"""

from __future__ import annotations

import re
from typing import Any

SCRIPT = re.compile(r"<script[^>]*>(.*?)</script>", re.S)
WRAPPER = re.compile(r"\(\$R=>")
CALL = re.compile(r"\$R\[\d+\]\.(next|return|throw)\(")
INDEX = re.compile(r"\d+")
KEY = re.compile(r"[A-Za-z_$][\w$]*|\d+")
NUMBER = re.compile(r"-?(?:\d+\.?\d*|\.\d+)(?:[eE][+-]?\d+)?")
HEX = re.compile(r"[0-9a-fA-F]+")
ESCAPES = {"n": "\n", "t": "\t", "r": "\r", "b": "\b", "f": "\f", "v": "\v", "0": "\0"}
WORDS: tuple[tuple[str, Any], ...] = (
    ("void 0", None),
    ("undefined", None),
    ("null", None),
    ("true", True),
    ("false", False),
    ("!0", True),
    ("!1", False),
)
MAX_DEPTH = 200


class _Unsupported(Exception):
    """The wrapper holds something outside the decoded subset."""


class _Ref:
    __slots__ = ("index",)

    def __init__(self, index: int) -> None:
        self.index = index


class _Parser:
    def __init__(self, text: str, pos: int) -> None:
        self.text, self.pos = text, pos
        self.table: dict[int, Any] = {}

    def take(self, token: str) -> bool:
        if self.text.startswith(token, self.pos):
            self.pos += len(token)
            return True
        return False

    def expect(self, token: str) -> None:
        if not self.take(token):
            raise _Unsupported

    def match(self, pattern: re.Pattern[str]) -> str:
        found = pattern.match(self.text, self.pos)
        if found is None:
            raise _Unsupported
        self.pos = found.end()
        return found.group()

    def value(self, depth: int = 0) -> Any:
        if depth > MAX_DEPTH or self.pos >= len(self.text):
            raise _Unsupported
        char = self.text[self.pos]
        if char == "{":
            return self.object(depth)
        if char == "[":
            return self.array(depth)
        if char == '"':
            return self.string()
        for word, decoded in WORDS:
            if self.take(word):
                return decoded
        if self.take("$R["):
            index = int(self.match(INDEX))
            self.expect("]")
            if self.take("="):
                self.table[index] = self.value(depth + 1)
                return self.table[index]
            return _Ref(index)
        number = self.match(NUMBER)
        return float(number) if any(c in number for c in ".eE") else int(number)

    def object(self, depth: int) -> dict[str, Any]:
        self.expect("{")
        out: dict[str, Any] = {}
        if self.take("}"):
            return out
        while True:
            key = self.string() if self.text.startswith('"', self.pos) else self.match(KEY)
            self.expect(":")
            out[key] = self.value(depth + 1)
            if self.take("}"):
                return out
            self.expect(",")

    def array(self, depth: int) -> list[Any]:
        self.expect("[")
        out: list[Any] = []
        if self.take("]"):
            return out
        while True:
            out.append(self.value(depth + 1))
            if self.take("]"):
                return out
            self.expect(",")

    def string(self) -> str:
        self.expect('"')
        chars: list[str] = []
        while self.pos < len(self.text):
            char = self.text[self.pos]
            self.pos += 1
            if char == '"':
                # `😀` arrives as two halves; re-encoding joins the pair.
                return "".join(chars).encode("utf-16", "surrogatepass").decode("utf-16")
            if char != "\\":
                chars.append(char)
                continue
            code = self.text[self.pos : self.pos + 1]
            self.pos += 1
            if code in ("x", "u"):
                width = 2 if code == "x" else 4
                digits = self.text[self.pos : self.pos + width]
                if len(digits) != width or HEX.fullmatch(digits) is None:
                    raise _Unsupported
                chars.append(chr(int(digits, 16)))
                self.pos += width
            else:
                chars.append(ESCAPES.get(code, code))
        raise _Unsupported


def _resolve(value: Any, table: dict[int, Any], seen: set[int]) -> Any:
    if isinstance(value, _Ref):
        target = table.get(value.index)
        return None if isinstance(target, _Ref) else _resolve(target, table, seen)
    if id(value) in seen:
        return value
    if isinstance(value, dict):
        seen.add(id(value))
        for key, item in value.items():
            value[key] = _resolve(item, table, seen)
    elif isinstance(value, list):
        seen.add(id(value))
        for index, item in enumerate(value):
            value[index] = _resolve(item, table, seen)
    return value


def _wrapper(body: str, start: int) -> tuple[Any, str, dict[int, Any]] | None:
    """Decode one `($R=>...)($R["key"])` wrapper; None for a non-data or unknown one."""
    parser = _Parser(body, start)
    call = CALL.match(body, start)
    if call is not None:
        if call.group(1) != "next":
            return None
        parser.pos = call.end()
    value = parser.value()
    parser.expect(")")
    if call is not None:
        parser.expect(")")
    parser.expect("($R[")
    key = parser.string()
    parser.expect("])")
    return value, key, parser.table


def decode_scripts(html: str) -> list[Any]:
    """Every data value the page streams, references resolved per stream; never raises."""
    tables: dict[str, dict[int, Any]] = {}
    decoded: list[tuple[Any, str]] = []
    for body in SCRIPT.findall(html):
        if "$R" not in body:
            continue
        for wrapper in WRAPPER.finditer(body):
            try:
                found = _wrapper(body, wrapper.end())
            except (_Unsupported, ValueError, RecursionError):
                continue
            if found is None:
                continue
            value, key, table = found
            tables.setdefault(key, {}).update(table)
            decoded.append((value, key))
    return [_resolve(value, tables[key], set()) for value, key in decoded]
```

"""SSRF-safe X URL identity and pure parsing of X's anonymous data.

The core of a post is the public syndication JSON
(`cdn.syndication.twimg.com/tweet-result`). The page's seroval-decoded data adds the
first-page replies, views, and reposts; it is an enrichment, so nothing missing there is an
error. Nothing here performs I/O.
"""

from __future__ import annotations

import re
from collections.abc import Iterable, Iterator, Sequence
from datetime import UTC, datetime
from typing import Any

from pydantic import Field

from thoth_control_plane.acquisition.adapters.threads import _https, _match_safe_path
from thoth_control_plane.acquisition.models import (
    MAX_COMMENTS,
    MAX_MEDIA_ITEMS,
    MAX_POST_TEXT,
    ReportComment,
    ResolvedMedia,
    SourcePost,
)
from thoth_control_plane.acquisition.normalize import _count, published_at_from_epoch
from thoth_control_plane.domain.models import StrictModel

CANONICAL_BASE = "https://x.com"
POST_PATH = re.compile(
    r"^/(?:[A-Za-z0-9_]{1,15}/status(?:es)?|i/web/status|i/status)/(?P<id>[0-9]{1,19})"
    r"(?:/(?:photo|video)/[1-4])?/?$"
)
HANDLE = re.compile(r"[A-Za-z0-9_]{1,15}")
SIZE = re.compile(r"/(\d{2,5})x(\d{2,5})/")
LEADING_MENTIONS = re.compile(r"^(?:@[A-Za-z0-9_]{1,15}\s+)+")
TRAILING_LINKS = re.compile(r"(?:\s*https://t\.co/[A-Za-z0-9]+)+\s*$")
MAX_COMMENT_TEXT = 2_000
VIDEO_TYPES = ("video", "animated_gif")


class XUrlError(ValueError):
    """Raised when a URL is not a safe, supported public X post URL."""


class XPostIdentity(StrictModel):
    post_id: str


class XParse(StrictModel):
    post: SourcePost
    media: list[ResolvedMedia]
    # A long tweet: the syndication text is cut short; the page's text of the post wins.
    long_text: bool = False


class XPageParse(StrictModel):
    engagement: dict[str, int] = Field(default_factory=dict)
    text: str | None = None
    comments: list[ReportComment] = Field(default_factory=list)
    dropped_comments: int = 0
    videos: list[str] = Field(default_factory=list)


def canonicalize_x_post_url(url: str) -> XPostIdentity:
    """Reduce any accepted X post URL to its numeric id; the author comes from the data."""
    return XPostIdentity(post_id=_match_safe_path(url, POST_PATH, "x", XUrlError).group("id"))


def post_url(handle: str, post_id: str) -> str:
    return f"{CANONICAL_BASE}/{handle}/status/{post_id}"


def _get(node: object, *keys: str) -> object:
    for key in keys:
        if not isinstance(node, dict):
            return None
        node = node.get(key)
    return node


def _handle(value: object) -> str | None:
    return value if isinstance(value, str) and HANDLE.fullmatch(value) else None


def _best_mp4(variants: object, url_key: str, type_key: str) -> str | None:
    """The largest-resolution https MP4; the first wins a tie or when no size is named."""
    best: tuple[int, str] | None = None
    for variant in variants if isinstance(variants, list) else []:
        if not isinstance(variant, dict) or variant.get(type_key) != "video/mp4":
            continue
        url = _https(variant.get(url_key))
        if url is None:
            continue
        size = SIZE.search(url)
        area = int(size.group(1)) * int(size.group(2)) if size else 0
        if best is None or area > best[0]:
            best = (area, url)
    return best[1] if best else None


def _published(value: object) -> str | None:
    if not isinstance(value, str):
        return None
    try:
        moment = datetime.fromisoformat(value)
    except ValueError:
        return None
    if moment.tzinfo is None:
        return None
    return moment.astimezone(UTC).strftime("%Y-%m-%dT%H:%M:%SZ")


def syndication_author(data: object) -> str | None:
    return _handle(_get(data, "user", "screen_name"))


def _syndication_media(data: dict[str, Any], page_videos: Sequence[str]) -> list[ResolvedMedia]:
    details = data.get("mediaDetails")
    listed = details if isinstance(details, list) else []
    items = [item for item in listed if isinstance(item, dict)]
    videos = [item for item in items if item.get("type") in VIDEO_TYPES]
    lone = _best_mp4(_get(data, "video", "variants"), "src", "type") if len(videos) == 1 else None
    media: list[ResolvedMedia] = []
    seen_videos = 0
    for item in items:
        if item.get("type") == "photo":
            url = _https(item.get("media_url_https"))
            if url is not None:
                original = url if "?" in url else f"{url}?name=orig"
                media.append(ResolvedMedia(kind="image", ephemeral_url=original, media_type=None))
        elif item.get("type") in VIDEO_TYPES:
            paged = page_videos[seen_videos] if seen_videos < len(page_videos) else None
            seen_videos += 1
            url = paged or lone
            if url is not None:
                media.append(ResolvedMedia(kind="video", ephemeral_url=url))
    return media[:MAX_MEDIA_ITEMS]


def parse_syndication(
    data: object, identity: XPostIdentity, page_videos: Sequence[str] = ()
) -> XParse | None:
    """Parse the tweet named by `identity`; None for a tombstone, another id, or a bad author."""
    if not isinstance(data, dict) or data.get("__typename") != "Tweet":
        return None
    if data.get("id_str") != identity.post_id:
        return None
    author = syndication_author(data)
    if author is None:
        return None
    text = data.get("text") if isinstance(data.get("text"), str) else ""
    span = data.get("display_text_range")
    if (
        isinstance(span, list)
        and len(span) == 2
        and all(isinstance(i, int) and not isinstance(i, bool) for i in span)
    ):
        text = text[span[0] : span[1]]
    engagement = {
        key: count
        for key, name in (("like", "favorite_count"), ("comment", "conversation_count"))
        if (count := _count(data.get(name))) is not None
    }
    return XParse(
        post=SourcePost(
            post_id=identity.post_id,
            owner_handle=author,
            text=text[:MAX_POST_TEXT],
            published_at=_published(data.get("created_at")),
            engagement=engagement,
        ),
        media=_syndication_media(data, page_videos),
        long_text=isinstance(data.get("note_tweet"), dict),
    )


def _walk(roots: Iterable[object]) -> Iterator[dict[str, Any]]:
    """Every dict in document order, each once; decoded seroval data can be cyclic."""
    stack: list[object] = list(reversed(list(roots)))
    seen: set[int] = set()
    while stack:
        value = stack.pop()
        if not isinstance(value, (dict, list)) or id(value) in seen:
            continue
        seen.add(id(value))
        if isinstance(value, dict):
            yield value
            stack.extend(reversed(list(value.values())))
        else:
            stack.extend(reversed(value))


def page_tweets(roots: Iterable[object]) -> list[dict[str, Any]]:
    """Each distinct `Tweet` in document order; X repeats a tweet across timeline entries."""
    tweets: list[dict[str, Any]] = []
    seen: set[str] = set()
    for node in _walk(roots):
        rest_id = node.get("rest_id")
        if node.get("__typename") == "Tweet" and isinstance(rest_id, str) and rest_id not in seen:
            seen.add(rest_id)
            tweets.append(node)
    return tweets


def author_of(tweet: dict[str, Any]) -> str | None:
    return _handle(_get(tweet, "core", "user_results", "result", "core", "screen_name"))


def display_text(text: str) -> str:
    """A tweet's own words: no leading reply mentions, no trailing `t.co` links."""
    return TRAILING_LINKS.sub("", LEADING_MENTIONS.sub("", text)).strip()


def tweet_text(tweet: dict[str, Any]) -> str | None:
    """A long tweet's note text, else its full text, without mentions and media links."""
    note = _get(tweet, "note_tweet", "note_tweet_results", "result", "text")
    text = note if isinstance(note, str) else _get(tweet, "details", "full_text")
    return display_text(text) if isinstance(text, str) else None


def tweet_published(tweet: dict[str, Any]) -> str | None:
    millis = _get(tweet, "details", "created_at_ms")
    if isinstance(millis, bool) or not isinstance(millis, int):
        return None
    return published_at_from_epoch(millis // 1000)


def tweet_engagement(tweet: dict[str, Any]) -> dict[str, int]:
    pairs = (
        ("like", _get(tweet, "counts", "favorite_count")),
        ("comment", _get(tweet, "counts", "reply_count")),
        ("share", _get(tweet, "counts", "retweet_count")),
        ("play", _get(tweet, "views", "count")),
    )
    return {key: count for key, value in pairs if (count := _count(value)) is not None}


def tweet_videos(tweet: dict[str, Any]) -> list[str]:
    """The best MP4 of each video entity, in order."""
    entities = tweet.get("media_entities2")
    videos: list[str] = []
    for entity in entities if isinstance(entities, list) else []:
        if isinstance(entity, dict) and entity.get("type") in VIDEO_TYPES:
            url = _best_mp4(_get(entity, "video_info", "variants"), "url", "content_type")
            if url is not None:
                videos.append(url)
    return videos


def _comment(tweet: dict[str, Any]) -> ReportComment | None:
    author, text = author_of(tweet), tweet_text(tweet)
    if author is None or not text:
        return None
    try:
        return ReportComment(
            comment_id=tweet["rest_id"],
            author=author,
            text=text[:MAX_COMMENT_TEXT],
            likes=_count(_get(tweet, "counts", "favorite_count")) or 0,
        )
    except ValueError:
        return None


def parse_page(roots: Iterable[object], identity: XPostIdentity) -> XPageParse:
    """Replies to the post and its page-only counters; an empty parse when none are found."""
    tweets = page_tweets(roots)
    post = next((tweet for tweet in tweets if tweet["rest_id"] == identity.post_id), None)
    comments: list[ReportComment] = []
    dropped = 0
    for tweet in tweets:
        if tweet is post or _get(tweet, "reply_to_results", "rest_id") != identity.post_id:
            continue
        if len(comments) == MAX_COMMENTS:
            break
        comment = _comment(tweet)
        if comment is None:
            dropped += 1
            continue
        comments.append(comment)
    if post is None:
        return XPageParse(comments=comments, dropped_comments=dropped)
    return XPageParse(
        engagement=tweet_engagement(post),
        text=tweet_text(post),
        comments=comments,
        dropped_comments=dropped,
        videos=tweet_videos(post),
    )


def with_page(parsed: XParse, page: XPageParse) -> XParse:
    """Add the page's views and reposts; a long tweet takes the page's full text."""
    text = page.text if parsed.long_text and page.text else parsed.post.text
    extra = {key: value for key, value in page.engagement.items() if key in ("play", "share")}
    post = parsed.post.model_copy(
        update={"text": text[:MAX_POST_TEXT], "engagement": {**parsed.post.engagement, **extra}}
    )
    return parsed.model_copy(update={"post": post})

"""Best-effort headless cards shared by the Instagram and X acquirers.

One lazily opened anonymous stealthy session visits a post page (the post card and one card
per tagged comment row) and a profile page (the profile card). Each platform supplies a
`CardLayout`. Every failure lands in `Cards.problems`; only cancellation propagates. A
card is always a real crop, never a synthetic render.
"""

from __future__ import annotations

import contextlib
import logging
from collections.abc import Awaitable, Callable
from dataclasses import asdict, dataclass, field
from pathlib import Path, PurePosixPath
from typing import Any
from urllib.parse import urlsplit

from thoth_control_plane.acquisition.browser import (
    FETCH_OPTIONS,
    HeadlessBrowserError,
    _classify_fetch_error,
    _default_session_factory,
    _increment_active_sessions,
)
from thoth_control_plane.acquisition.models import (
    AcquisitionReason,
    CardArtifact,
    ReportComment,
)
from thoth_control_plane.acquisition.threads_service import exit_session, save_card

logger = logging.getLogger(__name__)

COMMENT_ROW_ATTRIBUTE = "data-thoth-comment"
COMMENT_PREFIX_CHARS = 30
DIALOG_TIMEOUT_MS = 3_000


@dataclass(frozen=True)
class CardLayout:
    """Where one platform's cards live on its logged-out pages."""

    post_selector: str
    profile_selector: str
    # Called with the comment targets; tags rows with COMMENT_ROW_ATTRIBUTE, returns ids.
    tag_js: str
    login_paths: tuple[str, ...]
    dismiss_selector: str | None = None


def is_login_path(url: object, login_paths: tuple[str, ...]) -> bool:
    path = urlsplit(str(url)).path
    return any(path == prefix or path.startswith(prefix + "/") for prefix in login_paths)


@dataclass(frozen=True)
class CommentTarget:
    comment_id: str
    author: str
    prefix: str


def comment_targets(comments: list[ReportComment]) -> list[CommentTarget]:
    targets: list[CommentTarget] = []
    for comment in comments:
        line = next((part.strip() for part in comment.text.splitlines() if part.strip()), "")
        targets.append(
            CommentTarget(comment.comment_id, comment.author, line[:COMMENT_PREFIX_CHARS])
        )
    return targets


@dataclass
class Cards:
    post_png: bytes | None = None
    comment_pngs: dict[str, bytes] = field(default_factory=dict)
    profile_png: bytes | None = None
    problems: list[str] = field(default_factory=list)


async def _dismiss(page: Any, selector: str | None) -> None:
    # A sign-up dialog may cover the page; a missing dialog is fine.
    if selector is None:
        return
    with contextlib.suppress(Exception):
        await page.locator(selector).first.click(timeout=DIALOG_TIMEOUT_MS)


async def _crop(page: Any, selector: str) -> bytes | None:
    with contextlib.suppress(Exception):
        element = page.locator(selector).first
        await element.scroll_into_view_if_needed(timeout=5_000)
        return await element.screenshot(timeout=10_000)
    return None


def _on_post(
    cards: Cards, want_post: bool, targets: list[CommentTarget], layout: CardLayout
) -> Callable[[Any], Awaitable[None]]:
    async def action(page: Any) -> None:
        if is_login_path(page.url, layout.login_paths):
            cards.problems.append(AcquisitionReason.HEADLESS_BLOCKED.value)
            return
        await _dismiss(page, layout.dismiss_selector)
        if want_post:
            cards.post_png = await _crop(page, layout.post_selector)
        if not targets:
            return
        tagged: object = []
        with contextlib.suppress(Exception):
            tagged = await page.evaluate(layout.tag_js, [asdict(t) for t in targets])
        known = {target.comment_id for target in targets}
        for comment_id in tagged if isinstance(tagged, list) else []:
            # Only ids we sent (validated numeric) ever reach a selector.
            if comment_id in known and comment_id not in cards.comment_pngs:
                png = await _crop(page, f'[{COMMENT_ROW_ATTRIBUTE}="{comment_id}"]')
                if png is not None:
                    cards.comment_pngs[comment_id] = png

    return action


def _on_profile(cards: Cards, layout: CardLayout) -> Callable[[Any], Awaitable[None]]:
    async def action(page: Any) -> None:
        if is_login_path(page.url, layout.login_paths):
            cards.problems.append(AcquisitionReason.HEADLESS_BLOCKED.value)
            return
        await _dismiss(page, layout.dismiss_selector)
        cards.profile_png = await _crop(page, layout.profile_selector)

    return action


class CardBrowser:
    """One lazily opened stealthy session for a platform's card crops."""

    def __init__(
        self,
        layout: CardLayout,
        session_factory: Callable[..., Any] = _default_session_factory,
    ) -> None:
        self._layout = layout
        self._session_factory = session_factory
        self._session: Any | None = None
        self._active: Any | None = None

    async def _open(self) -> Any:
        if self._active is None:
            session = self._session_factory(headless=True, max_pages=1)
            try:
                active = await session.__aenter__()
            except Exception as error:
                raise HeadlessBrowserError(_classify_fetch_error(error)) from None
            self._session, self._active = session, active
            _increment_active_sessions()
        return self._active

    async def _visit(self, url: str, action: Callable[[Any], Awaitable[None]] | None = None):
        active = await self._open()
        try:
            return await active.fetch(url, page_action=action, **FETCH_OPTIONS)
        except Exception as error:
            raise HeadlessBrowserError(_classify_fetch_error(error)) from None

    async def _card_visit(
        self, url: str, action: Callable[[Any], Awaitable[None]], cards: Cards
    ) -> None:
        try:
            await self._visit(url, action)
        except HeadlessBrowserError as error:
            cards.problems.append(error.reason.value)

    async def cards(
        self,
        post_url: str,
        profile_url: str,
        *,
        want_post: bool,
        targets: list[CommentTarget],
    ) -> Cards:
        """Best effort: every failure lands in `problems`; only cancellation propagates."""
        cards = Cards()
        if want_post or targets:
            action = _on_post(cards, want_post, targets, self._layout)
            await self._card_visit(post_url, action, cards)
        await self._card_visit(profile_url, _on_profile(cards, self._layout), cards)
        return cards

    async def close(self) -> None:
        session, self._session, self._active = self._session, None, None
        await exit_session(session)


def persist_cards(
    cards: Cards, comments: list[ReportComment], workflow_id: str, artifact_root: Path
) -> tuple[CardArtifact | None, list[ReportComment], CardArtifact | None]:
    base = PurePosixPath("reports") / workflow_id / "cards"

    def save(png: bytes | None, name: str) -> CardArtifact | None:
        return save_card(png, artifact_root, base / name)

    carded = [
        comment.model_copy(
            update={
                "card": save(
                    cards.comment_pngs.get(comment.comment_id),
                    f"comment-{comment.comment_id}.png",
                )
            }
        )
        for comment in comments
    ]
    return save(cards.post_png, "post.png"), carded, save(cards.profile_png, "profile.png")


def log_outcome(
    platform: str,
    method: str,
    media: int,
    comments: list[ReportComment],
    post_card: CardArtifact | None,
    profile_card: CardArtifact | None,
    *,
    want_post: bool,
    problems: list[str],
    dropped: int,
) -> None:
    """Counts and enums only: never a URL, a handle, or text."""
    carded = sum(comment.card is not None for comment in comments)
    logger.info(
        "%s acquisition finished method=%s media=%d comments=%d comment_cards=%d "
        "post_card=%s profile_card=%s",
        platform,
        method,
        media,
        len(comments),
        carded,
        post_card is not None,
        profile_card is not None,
    )
    missing = []
    if want_post and post_card is None:
        missing.append("post")
    if profile_card is None:
        missing.append("profile")
    if carded < len(comments):
        missing.append(f"comments={len(comments) - carded}")
    if missing or problems:
        logger.warning(
            "%s cards missing missing=%s problems=%s",
            platform,
            ",".join(missing) or "none",
            ",".join(problems) or "none",
        )
    if dropped:
        logger.debug("%s comments dropped count=%d", platform, dropped)

class FakeRequest:
    def __init__(
        self,
        data: object = _SYND,
        page: str = PAGE,

    ) -> None:
        self.data = _synd() if data is _SYND else data

        self.calls: list[tuple[str, str]] = []

    async def syndication(self, post_id: str) -> object:
        self.calls.append(("syndication", post_id))
        if self.error is not None:

# Tags each matched reply row; targets arrive as the evaluate argument, never interpolated.
# A row is a top-level article (never a nested quote) whose own text mentions `@author` as a
# whole handle and holds the reply's text prefix.
TAG_REPLIES_JS = """
(targets) => {
  const tagged = [];
  const rows = Array.from(document.querySelectorAll("article")).filter(
    (article) => !article.parentElement || !article.parentElement.closest("article")
  );
  const own = (article) => {
    let text = article.innerText || "";
    for (const quote of article.querySelectorAll("article")) {
      text = text.replace(quote.innerText || "", " ");
    }
    return text;
  };
  const mentions = (text, handle) => {
    for (let at = text.indexOf(handle); at !== -1; at = text.indexOf(handle, at + 1)) {
      if (!/[a-z0-9_]/.test(text.charAt(at + handle.length))) return true;
    }
    return false;
  };
  for (const target of targets) {
    const handle = "@" + target.author.toLowerCase();
    const row = rows.find((article) => {
      const text = own(article);
      return !article.hasAttribute("data-thoth-comment")
        && mentions(text.toLowerCase(), handle)
        && text.includes(target.prefix);
    });
    if (!row) continue;
    row.setAttribute("data-thoth-comment", target.comment_id);
    tagged.push(target.comment_id);
  }
  return tagged;
}
"""

def _text(response: Any) -> str:
    body = getattr(response, "body", b"")
    return body.decode("utf-8", "replace") if isinstance(body, bytes) else str(body)


class XRequestFetcher:
    """Plain impersonated HTTP GETs: the syndication JSON and the post or profile page."""

    def __init__(self, get: Callable[..., Awaitable[Any]] | None = None) -> None:
        self._get = get or _scrapling_get

    async def _response(self, url: str, *, missing_ok: bool = False) -> Any | None:
        try:
            response = await self._get(url, **REQUEST_OPTIONS)
        except Exception as error:
            reason = (
                AcquisitionReason.REQUEST_TIMEOUT
                if _is_timeout(error)
                else AcquisitionReason.REQUEST_BLOCKED
            )
            raise RequestFetchError(reason) from None
        status = getattr(response, "status", None)
        if missing_ok and status == 404:
            return None
        if status != 200 or is_login_path(response.url, LOGIN_PATHS):
            raise RequestFetchError(AcquisitionReason.REQUEST_BLOCKED)
        return response

    async def syndication(self, post_id: str) -> object:
        """The tweet's JSON; None when X has no such tweet (404) or the body is not JSON."""
        response = await self._response(SYNDICATION_URL.format(post_id=post_id), missing_ok=True)
        if response is None:
            return None
        try:
            return json.loads(_text(response))
        except ValueError:
            return None

    async def page(self, url: str) -> str:
        return _text(await self._response(url))

class XAcquirer:
    """One X post: the syndication core, the page enrichment, media in order, and cards."""

    def __init__(
        self,
        request_fetcher: XRequestFetcher,
        headless: XHeadlessBrowser,
        materializer: MediaMaterializer,
    ) -> None:
        self._request = request_fetcher
        self._headless = headless
        self._materializer = materializer

    async def _page(self, url: str, identity: XPostIdentity) -> XPageParse:
        """Best effort: a failed fetch or an unreadable page gives an empty parse."""
        try:
            html = await self._request.page(url)
        except RequestFetchError as error:
            logger.warning("x page enrichment unavailable reason=%s", error.reason.value)
            return XPageParse()
        page = parse_page(decode_scripts(html), identity)
        if not page.engagement:
            logger.warning(
                "x page enrichment found no post replies=%d dropped=%d",
                len(page.comments),
                page.dropped_comments,
            )
        return page

    async def inspect(
        self, *, workflow_id: str, source_url: str, artifact_root: Path
    ) -> AcquisitionResult:
        strategy = _STRATEGIES[CONTENT_METHOD]
        try:
            try:
                identity = canonicalize_x_post_url(source_url)
            except XUrlError:
                return AcquisitionResult(
                    failure=AcquisitionFailure(code="invalid_source_url", retryable=False)
                )
            start = time.monotonic()
            reason: AcquisitionReason | None = None
            try:
                data = await self._request.syndication(identity.post_id)
            except RequestFetchError as error:
                data, reason = None, error.reason
            author = syndication_author(data)
            page = (
                await self._page(post_url(author, identity.post_id), identity)
                if author is not None
                else XPageParse()
            )
            parsed = parse_syndication(data, identity, page.videos)
            if parsed is None and reason is None:
                reason = AcquisitionReason.REQUEST_INCOMPLETE
            media: list[MaterializedMedia] = []
            if parsed is not None:
                parsed = with_page(parsed, page)
                if not parsed.post.text.strip() and not parsed.media:
                    attempt = _make_attempt(strategy, None, _elapsed_ms(start))
                    return AcquisitionResult(
                        failure=AcquisitionFailure(
                            code="no_content", retryable=False, attempts=[attempt]
                        )
                    )
                try:
                    media = await materialize_media(
                        self._materializer,
                        parsed.media,
                        f"x-{identity.post_id}",
                        workflow_id,
                        strategy,
                        artifact_root,
                    )
                except MediaMaterializationError:
                    reason = AcquisitionReason.MEDIA_VALIDATION_FAILED
            attempt = _make_attempt(strategy, reason, _elapsed_ms(start))
            if reason is not None or parsed is None:
                code = (reason or AcquisitionReason.REQUEST_INCOMPLETE).value
                return AcquisitionResult(
                    failure=AcquisitionFailure(
                        code=code, retryable=code in RETRYABLE_FAILURES, attempts=[attempt]
                    )
                )
            owner = parsed.post.owner_handle
            canonical = post_url(owner, identity.post_id)
            want_post = not any(item.kind == "video" for item in media)
            cards = await self._headless.cards(
                canonical,
                f"{CANONICAL_BASE}/{owner}",
                want_post=want_post,
                targets=comment_targets(page.comments),
            )
            post_card, comments, profile_card = await asyncio.to_thread(
                persist_cards, cards, page.comments, workflow_id, artifact_root
            )
            log_outcome(
                "x",
                CONTENT_METHOD,
                len(media),
                comments,
                post_card,
                profile_card,
                want_post=want_post,
                problems=cards.problems,
                dropped=page.dropped_comments,
            )
            return AcquisitionResult(
                report=SourceReport(
                    workflow_id=workflow_id,
                    source=ReportSource(platform="x", canonical_url=canonical),
                    post=parsed.post,
                    media=media,
                    outcome=AcquisitionOutcome(attempts=[attempt]),
                    comments=comments,
                    post_card=post_card,
                    profile_card=profile_card,
                )
            )
        finally:
            # Same shielded close as Instagram: a second cancellation must not interrupt
            # browser cleanup partway through.
            await asyncio.shield(self._headless.close())

Request only: one Scrapling GET of `/<handle>`, decoded from the page's seroval stream.
Anonymous pages serve about six recent tweets, so a profile often yields few videos; the
login feature that follows Stage 3d lengthens the timeline. Never raises; every failure is

def _repost(tweet: dict[str, Any]) -> bool:
    return isinstance(tweet.get("retweeted_status_results"), dict)


def _video(tweet: dict[str, Any], owner: str) -> DiscoveredVideo | None:
    try:
        return DiscoveredVideo(
            post_id=tweet["rest_id"],
            owner_handle=owner,
            canonical_url=post_url(owner, tweet["rest_id"]),
            caption=(tweet_text(tweet) or "")[:MAX_CAPTION],
            published_at=tweet_published(tweet),
            engagement=tweet_engagement(tweet),
        )
    except ValueError:
        return None


def parse_profile(roots: Iterable[object], handle: str, limit: int) -> DiscoveryResult:
    tweets = page_tweets(roots)
    if not tweets:
        return _empty("profile", DiscoveryReason.NO_DATA)
    videos: list[DiscoveredVideo] = []
    photo_posts = invalid = 0
    for tweet in tweets:
        owner = author_of(tweet)
        # Another author's tweet (a repost's original, a quote) is not the handle's post.
        if owner is None or owner.lower() != handle.lower() or _repost(tweet):
            invalid += 1
            continue
        if not tweet_videos(tweet):
            photo_posts += 1
            continue
        video = _video(tweet, owner)
        if video is None:
            invalid += 1
            continue
        videos.append(video)
    return DiscoveryResult(
        kind="profile",
        videos=rank_videos(videos)[:limit],
        pages=1,
        exhausted=False,
        drops=DiscoveryDrops(photo_posts=photo_posts, invalid=invalid),
    )


class XProfileDiscovery:
    def __init__(self, fetcher: XRequestFetcher | None = None) -> None:
        self._fetcher = fetcher or XRequestFetcher()

    async def discover(self, request: ProfileDiscovery) -> DiscoveryResult:
        result = await self._collect(request)
        reason = result.reason.value if result.reason is not None else "none"
        logger.info(
            "x discovery finished kind=profile videos=%d photo_posts=%d invalid=%d reason=%s",
            len(result.videos),
            result.drops.photo_posts,
            result.drops.invalid,
            reason,
        )
        if result.reason is not None:
            logger.warning("x discovery stopped early kind=profile reason=%s", reason)
        return result

    async def _collect(self, request: ProfileDiscovery) -> DiscoveryResult:
        try:
            html = await self._fetcher.page(f"{CANONICAL_BASE}/{request.handle}")
        except RequestFetchError as error:
            return _empty("profile", _reason(error))
        return parse_profile(decode_scripts(html), request.handle, request.limit)

"""Operator-run, read-only X capture for the Stage 3c evidence and golden fixtures.

Never run by the test suite and only after the operator authorizes the capture. Reads
`<kind> <url>` lines (kind: video, image, text, replies, profile) from `--urls` and writes
one raw JSON per line plus the card PNGs under `--out`. URLs, handles, and bodies are never
printed; stdout carries kinds, statuses, and counts only.

from __future__ import annotations

import argparse
import asyncio
import json
import logging
import re
import sys
from pathlib import Path
from typing import Any

from thoth_control_plane.acquisition.adapters.seroval import decode_scripts
from thoth_control_plane.acquisition.adapters.threads import _match_safe_path
from thoth_control_plane.acquisition.adapters.x import (
    CANONICAL_BASE,
    XUrlError,
    canonicalize_x_post_url,
    page_tweets,
    parse_page,
    post_url,
    syndication_author,
)
from thoth_control_plane.acquisition.card_browser import comment_targets
from thoth_control_plane.acquisition.threads_service import RequestFetchError
from thoth_control_plane.acquisition.x_service import XHeadlessBrowser, XRequestFetcher

KINDS = ("video", "image", "text", "replies", "profile")
SPACING_SECONDS = 5
PROFILE_PATH = re.compile(r"^/(?P<handle>[A-Za-z0-9_]{1,15})/?$")


async def _page(fetcher: XRequestFetcher, url: str) -> dict[str, Any]:
    try:
        return {"status": "ok", "html": await fetcher.page(url)}
    except RequestFetchError as error:
        return {"status": error.reason.value, "html": ""}


async def _post(kind: str, url: str, out: Path) -> dict[str, Any]:
    identity = canonicalize_x_post_url(url)
    fetcher = XRequestFetcher()
    raw: dict[str, Any] = {"kind": kind, "post_id": identity.post_id, "cards": None}
    try:
        data = await fetcher.syndication(identity.post_id)
        raw["syndication"] = {"status": "ok", "data": data}
    except RequestFetchError as error:
        raw["syndication"] = {"status": error.reason.value, "data": None}
    author = syndication_author(raw["syndication"]["data"])
    if author is None:
        return raw
    target = post_url(author, identity.post_id)
    await asyncio.sleep(SPACING_SECONDS)
    raw["page"] = await _page(fetcher, target)
    page = parse_page(decode_scripts(raw["page"]["html"]), identity)
    await asyncio.sleep(SPACING_SECONDS)
    browser = XHeadlessBrowser()
    try:
        cards = await browser.cards(
            target,
            f"{CANONICAL_BASE}/{author}",
            want_post=True,
            targets=comment_targets(page.comments),
        )
    finally:
        await browser.close()
    if cards.post_png is not None:
        (out / f"{kind}-post.png").write_bytes(cards.post_png)
    for n, png in enumerate(cards.comment_pngs.values(), start=1):
        (out / f"{kind}-comment-{n}.png").write_bytes(png)
    if cards.profile_png is not None:
        (out / f"{kind}-profile.png").write_bytes(cards.profile_png)
    raw["cards"] = {
        "post": cards.post_png is not None,
        "comments": len(cards.comment_pngs),
        "comment_targets": len(page.comments),
        "profile": cards.profile_png is not None,
        "problems": cards.problems,
    }
    return raw


async def _profile(kind: str, url: str, out: Path) -> dict[str, Any]:
    handle = _match_safe_path(url, PROFILE_PATH, "x", XUrlError).group("handle")
    raw: dict[str, Any] = {"kind": kind, "handle": handle}
    raw["page"] = await _page(XRequestFetcher(), f"{CANONICAL_BASE}/{handle}")
    return raw


def _summary(raw: dict[str, Any]) -> dict[str, Any]:
    summary: dict[str, Any] = {"kind": raw["kind"]}
    syndication = raw.get("syndication")
    if syndication is not None:
        data = syndication["data"]
        summary["syndication"] = {
            "status": syndication["status"],
            "typename": data.get("__typename") if isinstance(data, dict) else None,
        }
    page = raw.get("page")
    if page is not None:
        tweets = page_tweets(decode_scripts(page["html"]))
        summary["page"] = {"status": page["status"], "decoded_tweets": len(tweets)}
    if raw.get("cards") is not None:
        summary["cards"] = raw["cards"]
    return summary


async def _run(urls: Path, out: Path) -> int:
    out.mkdir(parents=True, exist_ok=True)
    for line in urls.read_text(encoding="utf-8").splitlines():
        if not line.strip():
            continue
        parts = line.split(maxsplit=1)
        if len(parts) != 2 or parts[0] not in KINDS:
            print(json.dumps({"error": "unknown_kind"}))
            return 2
        kind, url = parts
        capture = _profile if kind == "profile" else _post
        try:
            raw = await capture(kind, url.strip(), out)
        except XUrlError:
            print(json.dumps({"kind": kind, "error": "invalid_source_url"}))
            return 2
        (out / f"{kind}.json").write_text(json.dumps(raw), encoding="utf-8")
        print(json.dumps(_summary(raw)))
        await asyncio.sleep(SPACING_SECONDS)
    return 0


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--urls", type=Path, required=True)
    parser.add_argument("--out", type=Path, required=True)
    args = parser.parse_args(argv)
    # Scrapling logs fetched URLs at info level; keep them out of the operator's terminal.
    logging.getLogger("scrapling").setLevel(logging.WARNING)
    return asyncio.run(_run(args.urls, args.out))


if __name__ == "__main__":
    sys.exit(main())

Operator-run after an authorized capture. Only the keys the parsers read survive. Handles,
ids, and text are pseudonymized; media URLs are replaced, keeping their `<w>x<h>` segment;

Output reports counts only; raw values are never printed, and a fixture with any surviving
raw value is not written.
"""

import argparse
import json
import sys
from datetime import UTC, datetime
from pathlib import Path
from typing import Any

from thoth_control_plane.acquisition.adapters.seroval import decode_scripts
from thoth_control_plane.acquisition.adapters.x import (

from thoth_control_plane.acquisition.x_discovery import parse_profile
from thoth_control_plane.operations.threads_fixture import MEDIA_HOST, _leaves, leak_count
from thoth_control_plane.operations.tiktok_comment_fixture import _Pseudonyms
from thoth_control_plane.operations.tiktok_discovery_fixture import (
    SYNTHETIC_EPOCH,
    SYNTHETIC_EPOCH_STEP,
    _floor,
)

# Values the fixture keeps verbatim: type names, not identifying tokens.
ENUMS = frozenset(
    {
        "Tweet",
        "TweetTombstone",
        "photo",
        "video",
        "animated_gif",
        "video/mp4",
        "application/x-mpegURL",
    }
)
Fixture = tuple[dict[str, Any], dict[str, Any]]


def _without_enums(value: Any) -> Any:
    """The raw scope for the leak check, minus the enum strings the fixture keeps."""
    if isinstance(value, dict):
        return {key: _without_enums(item) for key, item in value.items()}
    if isinstance(value, list):
        return [_without_enums(item) for item in value]
    return None if isinstance(value, str) and value in ENUMS else value


class _Sanitizer:
    def __init__(self) -> None:
        self.names = _Pseudonyms()
        self.urls = 0
        self.epochs = 0

    def url(self, kind: str, original: object, extension: str) -> str:
        self.urls += 1
        size = SIZE.search(original) if isinstance(original, str) else None
        segment = f"/{size.group(1)}x{size.group(2)}" if size else ""
        return f"{MEDIA_HOST}/{kind}/{self.urls}{segment}/m.{extension}"

    def epoch(self) -> int:
        self.epochs += 1
        return SYNTHETIC_EPOCH + self.epochs * SYNTHETIC_EPOCH_STEP

    def counter(self, value: object) -> int | None:
        count = _count(value)
        return None if count is None else _floor(count)

    def syndication(self, data: dict[str, Any], alias: str | None) -> dict[str, Any]:
        out: dict[str, Any] = {
            "__typename": data.get("__typename"),
            "id_str": alias,
            "user": {"screen_name": self.names.handle(_get(data, "user", "screen_name"))},
        }
        if isinstance(data.get("text"), str):
            out["text"] = self.names.text(data["text"])
        span = data.get("display_text_range")
        if isinstance(span, list):
            out["display_text_range"] = span
        if data.get("created_at") is not None:
            moment = datetime.fromtimestamp(self.epoch(), tz=UTC)
            out["created_at"] = moment.strftime("%Y-%m-%dT%H:%M:%S.000Z")
        for key in ("favorite_count", "conversation_count"):
            if (count := self.counter(data.get(key))) is not None:
                out[key] = count
        details = data.get("mediaDetails")
        if isinstance(details, list):
            out["mediaDetails"] = [self._detail(item) for item in details if isinstance(item, dict)]
        variants = _get(data, "video", "variants")
        if isinstance(variants, list):
            out["video"] = {
                "variants": [
                    {"type": v.get("type"), "src": self.url("video", v.get("src"), "mp4")}
                    for v in variants
                    if isinstance(v, dict)
                ]
            }
        if isinstance(data.get("note_tweet"), dict):
            out["note_tweet"] = {"id": "note"}
        return out

    def _detail(self, item: dict[str, Any]) -> dict[str, Any]:
        out: dict[str, Any] = {"type": item.get("type")}
        if item.get("type") == "photo":
            out["media_url_https"] = self.url("image", item.get("media_url_https"), "jpg")
        return out

    def tweet(self, tweet: dict[str, Any]) -> dict[str, Any]:
        screen_name = _get(tweet, "core", "user_results", "result", "core", "screen_name")
        out: dict[str, Any] = {
            "__typename": "Tweet",
            "rest_id": self.names.comment_id(tweet.get("rest_id")),
            "core": {
                "user_results": {
                    "result": {"core": {"screen_name": self.names.handle(screen_name)}}
                }
            },
        }
        details: dict[str, Any] = {}
        text = _get(tweet, "details", "full_text")
        if isinstance(text, str):
            details["full_text"] = self.names.text(text)
        if _get(tweet, "details", "created_at_ms") is not None:
            details["created_at_ms"] = self.epoch() * 1000
        out["details"] = details
        note = _get(tweet, "note_tweet", "note_tweet_results", "result", "text")
        if isinstance(note, str):
            out["note_tweet"] = {"note_tweet_results": {"result": {"text": self.names.text(note)}}}
        counts = {
            key: count
            for key in ("favorite_count", "reply_count", "retweet_count")
            if (count := self.counter(_get(tweet, "counts", key))) is not None
        }
        out["counts"] = counts
        if (views := self.counter(_get(tweet, "views", "count"))) is not None:
            out["views"] = {"count": views}
        reply_to = _get(tweet, "reply_to_results", "rest_id")
        if reply_to is not None:
            out["reply_to_results"] = {"rest_id": self.names.comment_id(reply_to)}
        entities = tweet.get("media_entities2")
        if isinstance(entities, list):
            out["media_entities2"] = [self._entity(e) for e in entities if isinstance(e, dict)]
        if isinstance(tweet.get("retweeted_status_results"), dict):
            out["retweeted_status_results"] = {"result": {}}
        return out

    def _entity(self, entity: dict[str, Any]) -> dict[str, Any]:
        out: dict[str, Any] = {"type": entity.get("type")}
        variants = _get(entity, "video_info", "variants")
        if entity.get("type") in VIDEO_TYPES and isinstance(variants, list):
            out["video_info"] = {
                "variants": [
                    {
                        "content_type": v.get("content_type"),
                        "url": self.url("video", v.get("url"), "mp4"),
                    }
                    for v in variants
                    if isinstance(v, dict)
                ]
            }
        return out

def build_profile(raw: dict[str, Any]) -> Fixture:

    }
    return fixture, {"handle": raw["handle"], "tweets": tweets}


def build(raw: dict[str, Any]) -> Fixture:
    return build_profile(raw) if raw["kind"] == "profile" else build_post(raw)


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--raw", type=Path, required=True)
    parser.add_argument("--out", type=Path, required=True)
    args = parser.parse_args(argv)
    raw = json.loads(args.raw.read_text(encoding="utf-8"))
    fixture, scope = build(raw)
    # Compare raw string leaves with fixture string leaves, not JSON text: escaping in the
    # JSON text would hide a surviving value that contains a quote or a newline.
    leaks = leak_count(_without_enums(scope), "\n".join(sorted(_leaves(fixture))))
    print(
        json.dumps(
            {
                "kind": fixture["kind"],
                "method": fixture["method"],
                "counts": fixture["counts"],
                "leaks": leaks,
            }
        )
    )
    if leaks:
        return 1
    text = json.dumps(fixture, indent=2, ensure_ascii=False) + "\n"
    args.out.write_bytes(text.encode("utf-8"))
    return 0


if __name__ == "__main__":
    sys.exit(main())
